cmake_minimum_required(VERSION 3.16)
project(VilaCoreTests CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Debug)
endif()
file(GLOB CORE_SRC ${CMAKE_CURRENT_SOURCE_DIR}/../Private/*.cpp)
add_library(vilacore STATIC ${CORE_SRC})
target_include_directories(vilacore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/../Public)
target_compile_options(vilacore PRIVATE -Wall -Wno-unused-parameter)
add_executable(vila_tests ${CMAKE_CURRENT_SOURCE_DIR}/TestRunner.cpp)
target_link_libraries(vila_tests vilacore)
